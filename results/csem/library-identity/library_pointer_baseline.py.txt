from pathlib import Path
import subprocess,hashlib,json,os,sys
root=Path.cwd();out=root/'build/library-pointer-baseline';out.mkdir(exist_ok=False)
oracle,runtime=sys.argv[1:3];rows=[]
for name in ['redecl-pointer-neutral','redecl-pointer-shadow','pointer-neutral','pointer-shadow']:
 src=root/'build/library-focused-3/sources'/(name+'.c')
 cmd=[oracle,'--runtime='+runtime,'--nolibc','--exec','--batch','--mode=exhaustive',str(src)]
 p=subprocess.run(cmd,env=dict(os.environ,LEAN_ABORT_ON_PANIC='1'),capture_output=True,timeout=120)
 (out/(name+'.command.json')).write_text(json.dumps(cmd,indent=2)+'\n');(out/(name+'.stdout')).write_bytes(p.stdout);(out/(name+'.stderr')).write_bytes(p.stderr)
 exe=out/(name+'.exe');gcc=['gcc','-std=c11','-pedantic-errors','-fno-builtin','-O0',str(src),'-o',str(exe)]
 c=subprocess.run(gcc,capture_output=True);(out/(name+'-gcc.command.json')).write_text(json.dumps(gcc,indent=2)+'\n');(out/(name+'-gcc.stderr')).write_bytes(c.stderr)
 r=subprocess.run([exe],capture_output=True) if not c.returncode else None
 row=dict(case=name,source_sha256=hashlib.sha256(src.read_bytes()).hexdigest(),oracle_exit=p.returncode,stdout=p.stdout.decode(),stderr=p.stderr.decode(),gcc_compile_exit=c.returncode,gcc_exit=r.returncode if r else None)
 rows.append(row);print(name,'oracle',p.returncode,p.stdout.decode().strip(),'GCC',row['gcc_exit'])
(out/'results.json').write_text(json.dumps(dict(pins={'oracle':hashlib.sha256(Path(oracle).read_bytes()).hexdigest(),'csem_rev':(root/'vendor/CSEM_REV').read_text(),'gcc':subprocess.check_output(['gcc','--version'],text=True).splitlines()[0]},rows=rows),indent=2)+'\n')
